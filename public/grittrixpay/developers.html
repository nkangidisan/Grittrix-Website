Quick Start

1 Install SDK

npm 
npm install @grittrix/pay-sdk

2 Initialize

javascript
import GrittrixPay from '@grittrix/pay-sdk';

const pay = new GrittrixPay({
  apiKey: process.env.GRITTRIX_API_KEY,
  environment: 'production'
});

3 Create Payment

javascript
const payment = await pay.createPayment({
  amount: 120000,
  currency: 'UGX',
  phone: '+256772XXXXXX',
  network: 'auto',
  webhookUrl: 'https://yourapp.com/webhook'
});

console.log(payment.link); // → "https://pay.grittrix.com/p/abc"
